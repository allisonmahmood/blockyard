cmake_minimum_required(VERSION 3.24)
project(blockyard_ui_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick Qml QuickControls2 Test)
add_executable(blockyard_ui_test ui_test.cpp ../../src/treemap.cpp ../../src/treemap.h ../../src/desktop/theme.cpp ../../src/desktop/theme.h)
target_link_libraries(blockyard_ui_test PRIVATE Qt6::Quick Qt6::Qml Qt6::QuickControls2 Qt6::Test)
target_compile_definitions(blockyard_ui_test PRIVATE BLOCKYARD_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../..")
enable_testing()
add_test(NAME ui COMMAND blockyard_ui_test)
set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
