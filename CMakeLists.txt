cmake_minimum_required(VERSION 3.24)
project(Blockyard VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Concurrent Test)
qt_standard_project_setup(REQUIRES 6.4)
add_library(blockyard_engine STATIC src/engine/engine.cpp)
target_include_directories(blockyard_engine PUBLIC src)
target_link_libraries(blockyard_engine PUBLIC Qt6::Core)
add_library(blockyard_theme STATIC src/desktop/theme.cpp)
target_include_directories(blockyard_theme PUBLIC src)
target_link_libraries(blockyard_theme PUBLIC Qt6::Core Qt6::Gui)
qt_add_executable(blockyard src/main.cpp src/controller.cpp src/treemap.cpp)
target_include_directories(blockyard PRIVATE src)
target_link_libraries(blockyard PRIVATE blockyard_engine blockyard_theme Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent)
qt_add_resources(blockyard ui PREFIX "/" FILES qml/Main.qml assets/blockyard.svg)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    foreach(target blockyard blockyard_engine blockyard_theme)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endforeach()
endif()
if(BUILD_TESTING)
    add_executable(engine_test tests/engine_test.cpp)
    target_link_libraries(engine_test PRIVATE blockyard_engine Qt6::Test)
    add_test(NAME engine COMMAND engine_test)
    add_executable(theme_test tests/theme_test.cpp)
    target_link_libraries(theme_test PRIVATE blockyard_theme Qt6::Test)
    add_test(NAME theme COMMAND theme_test)
    set_tests_properties(theme PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_subdirectory(tests/qml)
    add_executable(controller_test tests/controller_test.cpp src/controller.cpp)
    target_link_libraries(controller_test PRIVATE blockyard_engine Qt6::Test Qt6::Gui Qt6::Concurrent)
    add_test(NAME controller COMMAND controller_test)
    add_executable(adversarial_test tests/adversarial_test.cpp src/controller.cpp)
    target_link_libraries(adversarial_test PRIVATE blockyard_engine Qt6::Test Qt6::Gui Qt6::Concurrent)
    add_test(NAME adversarial COMMAND adversarial_test)
    add_executable(scan_probe tests/scan_probe.cpp)
    target_link_libraries(scan_probe PRIVATE blockyard_engine)
endif()
include(GNUInstallDirs)
install(TARGETS blockyard RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES assets/com.blockyard.app.desktop DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/applications)
install(FILES assets/blockyard.svg DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/icons/hicolor/scalable/apps)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/licenses/blockyard)
